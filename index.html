function showProducts() { const list = document.getElementById("productsList"); if (!list) return; let products = JSON.parse(localStorage.getItem("products")) || []; list.innerHTML = ""; products.forEach(product => { list.innerHTML += `

${product.name}

Цена: ${product.price} ₽

Категория: ${product.category}

Город: ${product.city}

${product.description}

`; }); }
